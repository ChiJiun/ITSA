let books = JSON.parse(localStorage.books || '[]');
let nextId = Number(localStorage.nextId || 1);
let editId = null; // null = 新增模式；有值 = 正在更新哪一本

const $ = id => document.getElementById(id);
const esc = s => String(s).replace(/</g, '&lt;'); // 防止輸入 <script> 被當成 HTML 執行

function store() {
  localStorage.books = JSON.stringify(books);
  localStorage.nextId = nextId;
  render();
}

// (2) 查看：把陣列畫成表格
function render() {
  $('list').innerHTML = books.map(b => `
    <tr><td>${b.id}</td><td>${esc(b.title)}</td><td>${esc(b.author)}</td><td>${b.year}</td>
    <td><button onclick="del(${b.id})">刪除</button></td>
    <td><button onclick="edit(${b.id})">更新</button></td></tr>`).join('');
}

// (1) 添加 / (4) 更新：同一個表單，看 editId 決定
function save() {
  const b = { id: editId || nextId++, title: $('title').value, author: $('author').value, year: $('year').value };
  if (editId) books = books.map(x => x.id === editId ? b : x);
  else books.push(b);
  editId = null;
  $('btn').textContent = '添加圖書';
  document.forms[0].reset();
  store();
}

// (3) 刪除：filter 留下 id 不同的
function del(id) {
  books = books.filter(b => b.id !== id);
  store();
}

// (4) 按「更新」：把資料帶回表單，按鈕改成「儲存更新」
function edit(id) {
  const b = books.find(b => b.id === id);
  $('title').value = b.title;
  $('author').value = b.author;
  $('year').value = b.year;
  editId = id;
  $('btn').textContent = '儲存更新';
}

2024 P3｜健康度量管理系統（jQuery + PHP PDO + MySQL）

題目重點
  • 人員（醫檢員/受檢者）、醫檢項目、檢查結果三張表，分數 1~10
  • 醫檢員：註冊、登入、三張表新增/刪除/更新/查詢，同項目人員編號不重複
  • 受檢者：首次登入強制改密碼（12 碼、大小寫英文 + 數字），之後單頁看全部分數
  • 測試帳號：醫檢員 tech1/1234；受檢者 p1~p5/1234
解題思路
  • 一支 api.php 用 ?a=動作 分流；表名、欄位走白名單，SQL 一律 prepare
  • results 加 UNIQUE(user_id,item_id) → 資料庫層保證不重複；CHECK 限制 1~10
  • save 有 id 就 UPDATE、沒有就 INSERT，三張表共用同一段程式
  • 前端每列都是 input，改完按「存」；最後一列空白當新增列
  • 密碼規則前端 pattern 先擋、後端 preg_match 再驗一次

-- 健康度量管理系統：人員 / 醫檢項目 / 檢查結果 三張表（第三正規化）
SET NAMES utf8mb4;
CREATE DATABASE IF NOT EXISTS health DEFAULT CHARSET utf8mb4;
USE health;
DROP TABLE IF EXISTS results, items, users;

-- a. 人員：role 區分醫檢員(tech)/受檢者(patient)；must_change=1 代表第一次登入要強制改密碼
CREATE TABLE users (
  id INT AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(50) NOT NULL,
  account VARCHAR(50) NOT NULL UNIQUE,
  password VARCHAR(255) NOT NULL,
  role ENUM('tech','patient') NOT NULL,
  must_change TINYINT NOT NULL DEFAULT 1
);

-- b. 醫檢項目：度量分數尺規 1~10
CREATE TABLE items (
  id INT AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(50) NOT NULL,
  min_score TINYINT NOT NULL DEFAULT 1,
  max_score TINYINT NOT NULL DEFAULT 10
);

-- c. 檢查結果：誰(user_id) 的 哪一項(item_id) 幾分
--    UNIQUE(user_id,item_id) => 同一項目同一個人只會有一筆（人員編號不重複）
CREATE TABLE results (
  id INT AUTO_INCREMENT PRIMARY KEY,
  user_id INT NOT NULL,
  item_id INT NOT NULL,
  score TINYINT NOT NULL CHECK (score BETWEEN 1 AND 10),
  UNIQUE (user_id, item_id),
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  FOREIGN KEY (item_id) REFERENCES items(id) ON DELETE CASCADE
);

session_start();

$TABLES = [
  'users'   => ['name', 'account', 'password', 'role', 'must_change'],

function q($sql, $args = []) { global $db; $s = $db->prepare($sql); $s->execute($args); return $s; }

$a = $_GET['a'] ?? '';
$p = $_POST;
$t = $p['t'] ?? $_GET['t'] ?? '';

try {
  switch ($a) {
    case 'login':
      $u = q('SELECT * FROM users WHERE account=? AND password=?', [$p['account'], $p['password']])->fetch();

    case 'register': // 醫檢員自行註冊

        [$p['name'], $p['account'], $p['password']]);
      out(['msg' => '註冊成功，請登入']);

    case 'logout':
      session_destroy();
      out(['msg' => '已登出']);

    case 'chpwd': // c. 受檢者第一次登入改密碼：大小寫英文 + 數字混合，共 12 碼
      if (!isset($_SESSION['id'])) out(['msg' => '請先登入'], 401);
      if (!preg_match('/^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)[A-Za-z\d]{12}$/', $p['password']))
        out(['msg' => '密碼需為 12 碼，且包含大寫、小寫英文與數字'], 400);

      out(['msg' => '密碼已更新']);

    case 'mine': // d. 受檢者看自己的所有項目分數
      need('patient');
      if (q('SELECT must_change FROM users WHERE id=?', [$_SESSION['id']])->fetchColumn()) out(['msg' => '請先更新密碼'], 403);
      out(q('SELECT i.name, r.score FROM results r JOIN items i ON i.id=r.item_id WHERE r.user_id=?', [$_SESSION['id']])->fetchAll());

    case 'list': // b. 醫檢員查詢任一張表
      need('tech');
      if (!isset($TABLES[$t])) out(['msg' => '表名錯誤'], 400);
      out(q("SELECT * FROM $t ORDER BY id")->fetchAll());

    case 'save': // a. 新增（沒有 id）或更新（有 id）
      need('tech');
      if (!isset($TABLES[$t])) out(['msg' => '表名錯誤'], 400);
      $cols = $TABLES[$t];
      $vals = array_map(fn($c) => $p[$c] ?? null, $cols);
      if (empty($p['id'])) {
        q("INSERT INTO $t (" . implode(',', $cols) . ') VALUES (' . implode(',', array_fill(0, count($cols), '?')) . ')', $vals);
      } else {
        q("UPDATE $t SET " . implode('=?,', $cols) . '=? WHERE id=?', [...$vals, $p['id']]);

    case 'del': // a. 刪除

      out(['msg' => '已刪除']);
  }
  out(['msg' => '未知動作'], 400);

題目重點
  • 前台：選服務（一般 G / 開戶 A / 諮詢 C）取號，顯示號碼、目前叫號、等待清單，5 秒內自動更新
  • 櫃台：admin/password123 登入，下一位、完成服務、歷史紀錄、登出銷毀 session
  • 單一 FIFO 佇列；同時最多 1 筆服務中；違規回 409；服務非白名單回 400
  • 號碼 前綴+3 位數，超過 999 變 A1000，存 DB 不重置；重啟後完整還原
  • 執行：把 P1_Node離線套件.zip 的 node_modules 放進 p1-ans，npm start
解題思路
  • 用 Node 22 內建 node:sqlite（同步 API），不用 callback，也不會有併發交錯
  • 計數器一行：INSERT … ON CONFLICT DO UPDATE SET n=n+1 RETURNING n
  • 時間存毫秒數字排序最準，輸出時才格式化成 YYYY/M/D HH:mm:ss
  • session 自己做：Set 存 sid，登出 delete → 舊 Cookie 重放也是 401
  • 權限一定在伺服器檢查（auth middleware），前端停用按鈕只是輔助
  • 前端 element id 別取 history、login 這種和全域變數撞名的名字
