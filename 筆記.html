題目重點
  • canvas 500×500：橘框黃底大圓、藍色與紅色小圓，中心都在正中央
  • 依系統時間畫時針、分針、秒針；時針位置要把分與秒的進度算進去

解題思路
  • 角度用「一圈的比例 f」：x = O + len·sin(2πf)、y = O − len·cos(2πf)，0 在 12 點方向
  • 寫一個 line(f, from, to, width, color) 小函式，指針和刻度共用
  • 時針 f = (h%12 + m/60 + s/3600)/12；分針 f = (m + s/60)/60；秒針 f = s/60
  • canvas 後畫的蓋在上面：時針 → 分針 → 藍圓 → 紅圓 → 秒針
  • 數字用 fillText 搭配 textAlign、textBaseline 置中；setInterval(draw, 1000) 每秒重畫

const ctx = document.getElementById('c').getContext('2d');
const O = 250; // 圓心 x = y = 250（canvas 正中央）

// 角度用「一圈的比例」(0~1) 表示；0 在 12 點鐘方向，順時針增加
// 套用提示公式：x = O + len*sin(2π*f)，y = O - len*cos(2π*f)
const pos = (f, len) => [O + len * Math.sin(2 * Math.PI * f), O - len * Math.cos(2 * Math.PI * f)];

// 畫線：從半徑 from 到半徑 to（指針 from=0；刻度則是外框上的一小段）
function line(f, from, to, width, color) {
  ctx.beginPath();
  ctx.moveTo(...pos(f, from));
  ctx.lineTo(...pos(f, to));
  ctx.lineWidth = width;
  ctx.strokeStyle = color;
  ctx.stroke();
}

// 實心圓
function dot(r, color) {
  ctx.beginPath();
  ctx.arc(O, O, r, 0, 2 * Math.PI);
  ctx.fillStyle = color;
  ctx.fill();
}

function draw() {
  const d = new Date(), h = d.getHours(), m = d.getMinutes(), s = d.getSeconds();
  ctx.clearRect(0, 0, 500, 500);

  // 錶面：黃底、橘框（線寬 20）
  ctx.beginPath();
  ctx.arc(O, O, 240, 0, 2 * Math.PI);
  ctx.fillStyle = 'yellow';
  ctx.fill();
  ctx.lineWidth = 20;
  ctx.strokeStyle = 'orange';
  ctx.stroke();

  // 每個小時刻度：棕色短線（跨在橘框上）＋紅色數字，角度每次差 1/12 圈
  ctx.font = '24px Arial';
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillStyle = 'red';
  for (let i = 1; i <= 12; i++) {
    line(i / 12, 230, 250, 4, 'saddlebrown');
    ctx.fillText(i, ...pos(i / 12, 200));
  }

  // 數字時間：補零成 20 : 28 : 08
  ctx.fillStyle = 'black';
  ctx.font = '26px Arial';
  ctx.fillText([h, m, s].map(n => String(n).padStart(2, '0')).join(' : '), O, 130);

  // 由下而上疊：時針 → 分針 → 藍圓 → 紅圓 → 秒針（後畫的蓋在上面）
  // 時針要加上分鐘與秒的進度，不然 3:59 時針會停在 3 的位置
  line(((h % 12) + m / 60 + s / 3600) / 12, 0, 120, 12, 'black');
  line((m + s / 60) / 60, 0, 170, 8, 'purple');
  dot(12, 'blue');
  dot(5, 'red');
  line(s / 60, 0, 200, 5, 'red');
}

  $ext = strtolower(pathinfo($f['name'], PATHINFO_EXTENSION));

    out(['msg' => '上傳檔案格式只能是 JPG、JPEG、PNG 或 GIF'], 400);

}

$id = $_POST['id'] ?? 0;

const MAIL = /^[a-z0-9]([a-z0-9_.]*[a-z0-9])?@[a-z0-9-]+(\.[a-z0-9-]+)+$/i;

題目重點
  • 前台：選服務（一般 G / 開戶 A / 諮詢 C）取號，顯示號碼、目前叫號、等待清單，5 秒內自動更新
  • 櫃台：admin/password123 登入，下一位、完成服務、歷史紀錄、登出銷毀 session
  • 單一 FIFO 佇列；同時最多 1 筆服務中；違規回 409；服務非白名單回 400
  • 號碼 前綴+3 位數，超過 999 變 A1000，存 DB 不重置；重啟後完整還原
  • 執行：把 P1_Node離線套件.zip 的 node_modules 放進 p1-ans，npm start
解題思路
  • 用 Node 22 內建 node:sqlite（同步 API），不用 callback，也不會有併發交錯
  • 計數器一行：INSERT … ON CONFLICT DO UPDATE SET n=n+1 RETURNING n
  • 時間存毫秒數字排序最準，輸出時才格式化成 YYYY/M/D HH:mm:ss
  • session 自己做：Set 存 sid，登出 delete → 舊 Cookie 重放也是 401
  • 權限一定在伺服器檢查（auth middleware），前端停用按鈕只是輔助
  • 前端 element id 別取 history、login 這種和全域變數撞名的名字

const express = require('express');
const crypto = require('crypto');
const { DatabaseSync } = require('node:sqlite');

const db = new DatabaseSync(__dirname + '/queue.db');
db.exec(`
  CREATE TABLE IF NOT EXISTS counters (prefix TEXT PRIMARY KEY, n INTEGER NOT NULL);
  CREATE TABLE IF NOT EXISTS tickets (
    id INTEGER PRIMARY KEY AUTOINCREMENT,      -- 越早取號 id 越小
    number TEXT NOT NULL UNIQUE,               -- A001、G012…
    service TEXT NOT NULL,                     -- G / A / C
    status TEXT NOT NULL DEFAULT 'waiting',    -- waiting → serving → completed
    created_at INTEGER NOT NULL,               -- 時間存毫秒數字，排序最準
    called_at INTEGER,
    finished_at INTEGER
  );
  CREATE INDEX IF NOT EXISTS idx_status ON tickets(status, created_at);
`);

const sessions = new Set(); // 已登入的 session id（登出就刪掉）

const fmt = t => {
  if (!t) return null;
  const d = new Date(t); // 格式：2026/7/7 15:10:14
  return `${d.getFullYear()}/${d.getMonth() + 1}/${d.getDate()} ${p2(d.getHours())}:${p2(d.getMinutes())}:${p2(d.getSeconds())}`;
};
const view = t => !t ? null : { // 查無資料回傳 null（JSON 才會有 current: null）
  number: t.number, service: SERVICES[t.service],
  createdAt: fmt(t.created_at), calledAt: fmt(t.called_at), finishedAt: fmt(t.finished_at),
};
const all = (sql, ...a) => db.prepare(sql).all(...a).map(view);
const current = () => view(db.prepare("SELECT * FROM tickets WHERE status='serving'").get());

app.post('/api/tickets', (req, res) => {
  const s = req.body.service;
  // 接受代碼(G/A/C)或中文名稱；不在白名單 → 400
  // 用 Object.hasOwn 而不是 SERVICES[s]：否則 'toString'、'__proto__' 會被當成合法服務
  // typeof 檢查：送陣列 ["G"] 時 hasOwn 也會判成 true，要先擋掉
  const code = typeof s !== 'string' ? null : Object.hasOwn(SERVICES, s) ? s : Object.keys(SERVICES).find(k => SERVICES[k] === s);
  if (!code) return res.status(400).json({ error: '服務類別錯誤' });
  // 計數器 +1 後取回（不存在就從 1 開始），序號存在 DB → 重開機也接續
  const { n } = db.prepare('INSERT INTO counters VALUES (?, 1) ON CONFLICT(prefix) DO UPDATE SET n = n + 1 RETURNING n').get(code);
  const number = code + String(n).padStart(3, '0'); // 1000 以上自然變 4 位數
